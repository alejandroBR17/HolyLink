      {shouldRotate && (
        <div className="absolute top-6 left-6 z-50 bg-black/95 text-white border border-stone-800 rounded-xl px-5 py-3 text-sm flex items-center gap-2 font-medium pointer-events-none tracking-wide shadow-xl backdrop-blur-md animate-pulse">
          <Tv className="w-5 h-5 text-yellow-500" />
          <span>📺 Modo Widescreen Bloqueado. Deite a tela.</span>
        </div>
      )}

      {/* ==================== SETTINGS & ALERTS ==================== */}
      <AnimatePresence>
        {isSettingsOpen && (
          <motion.div
            initial={{ opacity: 0, y: 50 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: 50 }}
            className="absolute bottom-6 right-6 z-[100] bg-stone-900 border border-stone-800 rounded-2xl p-6 shadow-2xl w-[450px]"
          >
            <div className="flex items-center justify-between mb-6">
              <div className="flex items-center gap-3">
                <Settings className="text-stone-400 w-5 h-5" />
                <h3 className="text-white font-bold text-lg">Configurações</h3>
              </div>
              <button onClick={() => setIsSettingsOpen(false)} className="text-stone-500 hover:text-white transition-colors">
                <X className="w-5 h-5" />
              </button>
            </div>
            
            <div className="space-y-6">
              <div>
                <label className="text-stone-400 text-sm font-medium mb-3 block">Alerta Visual na Tela</label>
                <div className="grid grid-cols-2 gap-3">
                  <button 
                    onClick={() => setActiveAlert(activeAlert === 'baby' ? null : 'baby')}
                    className={`p-4 rounded-xl border text-sm font-bold transition-colors ${activeAlert === 'baby' ? 'bg-yellow-500 text-black border-yellow-500' : 'bg-stone-800 border-stone-700 text-stone-300 hover:bg-stone-700'}`}
                  >
                    Choro de Bebê
                  </button>
                  <button 
                    onClick={() => setActiveAlert(activeAlert === 'car' ? null : 'car')}
                    className={`p-4 rounded-xl border text-sm font-bold transition-colors ${activeAlert === 'car' ? 'bg-yellow-500 text-black border-yellow-500' : 'bg-stone-800 border-stone-700 text-stone-300 hover:bg-stone-700'}`}
                  >
                    Veículo / Estacionamento
                  </button>
                </div>
              </div>
            </div>
            
            <p className="text-stone-600 text-xs mt-6 text-center">Pressione ESC para fechar</p>
          </motion.div>
        )}
      </AnimatePresence>

      {/* VISUAL ALERT OVERLAY */}
      <AnimatePresence>
        {activeAlert && (
          <motion.div
            initial={{ opacity: 0, y: -100 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -100 }}
            className="absolute top-10 left-1/2 -translate-x-1/2 z-[90] bg-yellow-500 text-black px-10 py-6 rounded-3xl shadow-[0_0_50px_rgba(234,179,8,0.3)] flex items-center gap-6"
          >
            <Bell className="w-12 h-12 animate-[bounce_2s_infinite]" strokeWidth={2} />
            <div>
              <h2 className="font-black text-3xl uppercase tracking-widest">Atenção</h2>
              <p className="font-bold text-2xl mt-1">
                {activeAlert === 'baby' ? 'Atenção pais: Compareçam ao berçário (EBI).' : 'Proprietário de veículo: Compareça ao estacionamento.'}
              </p>
            </div>
            <button 
              onClick={() => setActiveAlert(null)}
              className="ml-6 bg-black/10 hover:bg-black/20 p-3 rounded-full transition-colors"
            >
              <X className="w-8 h-8" />
            </button>
          </motion.div>
        )}
      </AnimatePresence>

      {/* HIDDEN INVISIBLE BUTTON TO OPEN SETTINGS IN THE BOTTOM RIGHT CORNER */}
      <div 
         className="absolute bottom-0 right-0 w-32 h-32 z-50 cursor-pointer"
         onClick={() => setIsSettingsOpen(true)}
      />
